create database practice;

create table students
(
id int,
first_name varchar(30),
last_name varchar(30),
email varchar(30),
mobile varchar(18),
address varchar(100)
);

/* this creates a table with empty structure
 
in the above table I need to add some more columns so I use ALTER to modify the table 

these CREATE,ALTER,DROP,TRUNCATE are DDL Commands */

-- so now we use ALTER to modify the table 

alter table students add column class varchar(20);

--so this add an new column in the table

/* to insert values into this table we use DML commands like
INSERT → add data
UPDATE → modify data
DELETE → remove data
SELECT → retrieve data */

UPDATE students

    last_name = 'Walker',
    email = 'jane@gmail.com',
    mobile = '9977665544',
    city = 'Vizag'
WHERE id = 2;

-- here we are not updating class value so that it remains same 
-- to rename the column name we use RENAME

alter table students rename column address to city;

--The DELETE query in SQL is used to remove data (rows) from a table.

DELETE FROM students
WHERE id = 4;

--IF WE need to delete a record we can took a backup of that 
create table students_backup as
select first_name,last_name,email,mobile as phone,city
from students where id = 4;

--by using orderby clause we can sort data in ascending and descending formats

SELECT * FROM students
ORDER BY city ASC; -- this can sort data in ascending order 

SELECT * FROM students
ORDER BY id desc; -- this sort data in descending order based on student id
